# 이번 검토 대상: 환율 보정 (팀원 5명이 매일 만지는 경로)

## 동작 요약
- 요율표는 KRW 정적값이다. 현지통화 원가(호텔·식비·가이드·차량·관광)는 요율을 정한
  시점 이후 환율이 변하면 실제 원가와 벌어진다.
- 그래서 목적지별로 '기준환율'(rate_fx_baseline)을 두고, 견적 계산 시
  (오늘환율 / 기준환율) 비율을 그 항목들에 곱한다. ±30% 클램프.
- 오늘환율(fx_rates)은 매일 새벽 Vercel cron이 외부 API에서 받아 저장한다.
- **기준환율은 담당자가 그 목적지 요율을 저장할 때마다 '오늘 환율'로 재설정된다.**
  요율 저장 경로는 세 가지고 전부 rateDate를 오늘로 갱신한다:
    (a) 개별 편집(가격 변경)  (b) 권역 일괄 조정(%)  (c) "확인함" 버튼
        — (c)는 랜드사에 물어봤더니 가격 변동이 없더라, 는 경우에 누른다.
          가격은 그대로 두고 요율 기준월만 오늘로 갱신한다.

## 특히 봐 주셨으면 하는 점
- (c) "확인함"이 기준환율까지 재설정하는 게 맞습니까? 담당자 의도는 "가격 그대로"인데
  기준환율이 오늘로 리셋되면 그동안 쌓인 환율 보정이 사라집니다. 맞는 동작입니까,
  아니면 조용히 금액이 틀어지는 겁니까? **KRW 요율표라는 점을 감안해서** 판단해 주세요.
- 팀원 5명이 매주 여러 목적지를 갱신하기 시작하면 이 구조에서 무엇이 깨집니까?
- 클램프(±30%)에 걸린 상태가 지속되면 어떻게 됩니까? 걸린 사실이 어디에도 안 보입니다.

function getPeakInfo(startDateStr, destKey) {
  if (!startDateStr) return { factor: 1.0, label: '' };
  const d = new Date(startDateStr);
  if (isNaN(d.getTime())) return { factor: 1.0, label: '' };
  const mmdd = `${String(d.getMonth() + 1).padStart(2, '0')}-${String(d.getDate()).padStart(2, '0')}`;
  const ymd = `${d.getFullYear()}-${mmdd}`;
  const inRange = (from, to) => (from <= to ? (mmdd >= from && mmdd <= to) : (mmdd >= from || mmdd <= to));
  const applies = (p) => p.keys === 'ALL' || (Array.isArray(p.keys) && p.keys.includes(destKey));
  let best = { factor: 1.0, label: '' };
  for (const p of PEAK_CALENDAR) {
    if (applies(p) && inRange(p.from, p.to) && p.factor > best.factor) best = { factor: p.factor, label: p.label };
  }
  /* 음력 연휴는 연도별 절대 날짜라 문자열 비교만으로 끝난다(YYYY-MM-DD는 사전순=시간순).
     등록 안 된 연도는 어떤 구간에도 안 걸려 자연히 1.0이 된다 — 의도된 동작이다. */
  for (const p of LUNAR_PEAKS) {
    if (applies(p) && ymd >= p.from && ymd <= p.to && p.factor > best.factor) best = { factor: p.factor, label: p.label };
  }
  return best;
}

# 코드 C: admin.html — 관리자 화면 환율 변동 배지
  function adminGetFxDrift(destKey) {
    const currency = typeof DEST_CURRENCY !== 'undefined' ? DEST_CURRENCY[destKey] : null;
    if (!currency) return null;
    const baseline = fxBaselineCache[destKey];
    const current  = fxRatesCache[currency];
    if (!baseline || !baseline.rate || typeof current !== 'number') return null;
    const pct = Math.abs(current - baseline.rate) / baseline.rate * 100;
    return { pct, drift: pct >= FX_DRIFT_THRESHOLD_PCT };
  }

  function adminGetCombinedStatus(destKey, rateDate) {
    const rs = adminGetRateStatus(rateDate);
    const fx = adminGetFxDrift(destKey);
    if (fx && fx.drift && rs.status === 'ok') {
      return { status:'check', months: rs.months, label:'확인 권장', icon:'⚠️', fxPct: fx.pct };
    }
    return { ...rs, fxPct: (fx && fx.drift) ? fx.pct : null };
  }
